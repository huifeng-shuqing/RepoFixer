    61 | class SpooledIOBase(IOBase):
    62 |     """
    63 |     A base class shared by the SpooledBytesIO and SpooledStringIO classes.
    64 | 
    65 |     The SpooledTemporaryFile class is missing several attributes and methods
    66 |     present in the StringIO implementation. This brings the api as close to
    67 |     parity as possible so that classes derived from SpooledIOBase can be used
    68 |     as near drop-in replacements to save memory.
    69 |     """
    70 |     __metaclass__ = ABCMeta
    71 | 
    72 |     def __init__(self, max_size=5000000, dir=None):
    73 |         self._max_size = max_size
    74 |         self._dir = dir

    76 |     def _checkClosed(self, msg=None):
    77 |         """Raise a ValueError if file is closed"""
    78 |         if self.closed:
    79 |             raise ValueError('I/O operation on closed file.'
    80 |                              if msg is None else msg)

   170 |     def truncate(self, size=None):
   171 |         """
   172 |         Truncate the contents of the buffer.
   173 | 
   174 |         Custom version of truncate that takes either no arguments (like the
   175 |         real SpooledTemporaryFile) or a single argument that truncates the
   176 |         value to a certain index location.
   177 |         """
   178 |         self._checkClosed()
   179 |         if size is None:
   180 |             return self.buffer.truncate()
   181 | 
   182 |         if size < 0:
   183 |             raise OSError(EINVAL, "Negative size not allowed")
   184 | 
   185 |         # Emulate truncation to a particular location
   186 |         pos = self.tell()
   187 |         self.seek(size)
   188 |         self.buffer.truncate()
   189 |         if pos < size:
   190 |             self.seek(pos)

   281 | class SpooledBytesIO(SpooledIOBase):
   282 |     """
   283 |     SpooledBytesIO is a spooled file-like-object that only accepts bytes. On
   284 |     Python 2.x this means the 'str' type; on Python 3.x this means the 'bytes'
   285 |     type. Bytes are written in and retrieved exactly as given, but it will
   286 |     raise TypeErrors if something other than bytes are written.
   287 | 
   288 |     Example::
   289 | 
   290 |         >>> from boltons import ioutils
   291 |         >>> with ioutils.SpooledBytesIO() as f:
   292 |         ...     f.write(b"Happy IO")
   293 |         ...     _ = f.seek(0)
   294 |         ...     isinstance(f.getvalue(), bytes)
   295 |         True
   296 |     """
   297 | 
   298 |     def read(self, n=-1):
   299 |         self._checkClosed()
   300 |         return self.buffer.read(n)
   301 | 
   302 |     def write(self, s):
   303 |         self._checkClosed()
   304 |         if not isinstance(s, bytes):
   305 |             raise TypeError("bytes expected, got {}".format(
   306 |                 type(s).__name__
   307 |             ))
   308 | 
   309 |         if self.tell() + len(s) >= self._max_size:
   310 |             self.rollover()
   311 |         self.buffer.write(s)
   312 | 
   313 |     def seek(self, pos, mode=0):
   314 |         self._checkClosed()
   315 |         return self.buffer.seek(pos, mode)
   316 | 
   317 |     def readline(self, length=None):
   318 |         self._checkClosed()
   319 |         if length:
   320 |             return self.buffer.readline(length)
   321 |         else:
   322 |             return self.buffer.readline()
   323 | 
   324 |     def readlines(self, sizehint=0):
   325 |         return self.buffer.readlines(sizehint)
   326 | 
   327 |     def rollover(self):
   328 |         """Roll the StringIO over to a TempFile"""
   329 |         if not self._rolled:
   330 |             tmp = TemporaryFile(dir=self._dir)
   331 |             pos = self.buffer.tell()
   332 |             tmp.write(self.buffer.getvalue())
   333 |             tmp.seek(pos)
   334 |             self.buffer.close()
   335 |             self._buffer = tmp
   336 | 
   337 |     @property
   338 |     def _rolled(self):
   339 |         return not isinstance(self.buffer, BytesIO)
   340 | 
   341 |     @property
   342 |     def buffer(self):
   343 |         try:
   344 |             return self._buffer
   345 |         except AttributeError:
   346 |             self._buffer = BytesIO()
   347 |         return self._buffer
   348 | 
   349 |     @property
   350 |     def len(self):
   351 |         """Determine the length of the file"""
   352 |         pos = self.tell()
   353 |         if self._rolled:
   354 |             self.seek(0)
   355 |             val = os.fstat(self.fileno()).st_size
   356 |         else:
   357 |             self.seek(0, os.SEEK_END)
   358 |             val = self.tell()
   359 |         self.seek(pos)
   360 |         return val
   361 | 
   362 |     def tell(self):
   363 |         self._checkClosed()
   364 |         return self.buffer.tell()
