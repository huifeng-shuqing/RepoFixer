def _get_code_type(format_type: str):
    """Returns the type of country code."""
    if format_type.isdecimal():
        return "numeric"
    if format_type.isalpha():
        if len(format_type) == 2:
            return "alpha2"
        if len(format_type) == 3:
            return "alpha3"
    return "invalid"
