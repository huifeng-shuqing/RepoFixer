"""Test directory negations below an excluded ancestor."""

from unittest import TestCase
from unittest.mock import patch

from .test_06_gitignore import GitIgnoreSpecMixin


class AncestorNegationTest(GitIgnoreSpecMixin, TestCase):

	def test_excluded_ancestor(self):
		# Expected ignore decisions and pattern indices verified with
		# git check-ignore --no-index (Git 2.52.0.windows.1).
		for lines, paths in (
			(['a/', '!a/b/'], ['a/b/file.txt', 'a/b/c/deep.txt']),
			(['a/', '!a/b/', '!*.txt'], ['a/b/file.txt']),
			(['a/', '!a/b/', '*.txt'], ['a/b/file.txt']),
			(['d/**/', '!d/child/'], ['d/child/grand/deep.txt']),
			(['a/**', '!a/b/**'], ['a/b/file.txt', 'a/b/c/deep.txt']),
			(['a/**', '!*.txt'], ['a/b/file.txt']),
			(['*', '!*.txt'], ['a/top.txt', 'a/b/file.txt']),
		):
			for sub_test in self.parameterize_from_lines(lines):
				with sub_test() as spec:
					results = list(spec.check_files(iter(paths)))
					self.assertEqual(
						[(r.include, r.index) for r in results],
						[(True, 0)] * len(paths),
						(lines, results),
					)
					self.assertEqual(list(spec.match_files(iter(paths))), paths)
					self.assertEqual(list(spec.match_files(iter(paths), negate=True)), [])

	def test_reincluded_ancestors(self):
		for lines, path in (
			(['a/', '!a/', '!a/b/'], 'a/b/file.txt'),
			(['a/*', '!a/b/'], 'a/b/file.txt'),
			(['d/**/', '!d/child/'], 'd/child/top.txt'),
			(['d/**/', '!d/child/', '!d/child/grand/'], 'd/child/grand/deep.txt'),
			(['!a/b/'], 'a/b/file.txt'),
			(['a/**', '!*.txt'], 'a/top.txt'),
			(['*', '!a/', '!a/**'], 'a/b/file.txt'),
		):
			for sub_test in self.parameterize_from_lines(lines):
				with sub_test() as spec:
					self.assertFalse(spec.match_file(path), (lines, path))

	def test_d(self):
		"""
		Test terminal-globstar negation with excluded ancestor directories.
		"""
		for sub_test in self.parameterize_from_lines([
			"d/**/",
			"!d/child/",
		]):
			with sub_test() as spec:
				# PathSpec leaves the bare pattern-root directory unmatched so a
				# traversal can still reach its direct files.
				self.assertFalse(spec.match_file("d/"))
				self.assertFalse(spec.match_file("d/top.txt"))
				self.assertFalse(spec.match_file("d/child/"))
				self.assertFalse(spec.match_file("d/child/top.txt"))
				self.assertTrue(spec.match_file("d/child/grand/"))
				# Re-including child does not re-include the excluded grand directory.
				self.assertTrue(spec.match_file("d/child/grand/deep.txt"))

		for sub_test in self.parameterize_from_lines(["!d/**/"]):
			with sub_test() as spec:
				# The terminal-descendant rule applies to negated patterns too.
				for path in ("d/", "d/top.txt"):
					result = spec.check_file(path)
					self.assertIsNone(result.include)
					self.assertIsNone(result.index)
				for path in (
					"d/child/",
					"d/child/file.txt",
					"d/child/grand/deep.txt",
				):
					with self.subTest(negated_pattern_path=path):
						result = spec.check_file(path)
						self.assertFalse(result.include)
						self.assertEqual(result.index, 0)


class GitIgnoreSpecIssue139Test(GitIgnoreSpecMixin, TestCase):
	"""
	The :class:`GitIgnoreSpecIssue139Test` class tests the :class:`.GitIgnoreSpec`
	implementation for issue #139.
	"""

	def test_1(self):
		"""
		Test that glob-stars match newlines in names.
		"""
		for pattern, path in [
			("target", "line\nbreak/target"),
			("*/target", "line\nbreak/target"),
			("**/target", "line\nbreak/target"),
			("root/*/target", "root/line\nbreak/target"),
			("root/**/target", "root/line\nbreak/target"),
			("*/target", "\n/target"),
			("**/target", "\n/target"),
		]:
			for sub_test in self.parameterize_from_lines([pattern]):
				with sub_test() as spec:
					self.assertTrue(spec.match_file(path))
