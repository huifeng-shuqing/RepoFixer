                        with self.assertRaises(ValueError):
                            stream.seek(offset, whence)
                        self.assertEqual(stream.tell(), 2)
                        self.assertEqual(stream.read(), 'bc')

    def test_tell_codepoints(self):

        self.assertEqual(self.spooled_flo.tell(), 40)
