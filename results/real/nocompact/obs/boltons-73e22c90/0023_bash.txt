    def _calc_pearson_type(self):
        precision = self._pearson_precision
        skewness = self.skewness
        kurtosis = self.kurtosis
        beta1 = skewness ** 2.0
        beta2 = kurtosis * 1.0

        # TODO: range checks?

        c0 = (4 * beta2) - (3 * beta1)
        c1 = skewness * (beta2 + 3)
        c2 = (2 * beta2) - (3 * beta1) - 6

        if round(c1, precision) == 0:
            if round(beta2, precision) == 3:
                return 0  # Normal
            else:
                if beta2 < 3:
                    return 2  # Symmetric Beta
                elif beta2 > 3:
                    return 7
        elif round(c2, precision) == 0:
            return 3  # Gamma
        else:
            denom = 4 * c0 * c2
            if round(denom, precision) == 0:
                # Degenerate case: the classification criterion is
                # undefined, so fall back to the sign of the
                # discriminant-like term, c0.
                if c0 < 0:
                    return 1  # Beta
                elif c0 == 0:
                    return 0  # Normal
                else:
                    return 6  # Type VI
            k = c1 ** 2 / denom
            if k < 0:
                return 1  # Beta
            elif round(k, precision) == 1:
                return 5  # Type V
            elif k < 1:
                return 4  # Type IV
            else:
                return 6  # Type VI
        raise RuntimeError('missed a spot')
