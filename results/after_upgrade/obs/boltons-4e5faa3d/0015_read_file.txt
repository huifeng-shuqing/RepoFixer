   407 |     def _traverse_codepoints(self, current_position, n):
   408 |         """Traverse from current position to the right n codepoints"""
   409 |         dest = current_position + n
   410 |         while True:
   411 |             if current_position == dest:
   412 |                 # By chance we've landed on the right position, break
   413 |                 break
   414 | 
   415 |             # If the read would take us past the intended position then
   416 |             # seek only enough to cover the offset
   417 |             if current_position + READ_CHUNK_SIZE > dest:
   418 |                 self.read(dest - current_position)
   419 |                 break
   420 |             else:
   421 |                 ret = self.read(READ_CHUNK_SIZE)
   422 | 
   423 |             # Increment our current position
   424 |             current_position += READ_CHUNK_SIZE
   425 | 
   426 |             # If we kept reading but there was nothing here, break
   427 |             # as we are at the end of the file
   428 |             if not ret:
   429 |                 break
   430 | 
   431 |         return dest
   432 | 
   433 |     def seek(self, pos, mode=0):
   434 |         """Traverse from offset to the specified codepoint"""
   435 |         self._checkClosed()
   436 |         # Seek to position from the start of the file
   437 |         if mode == os.SEEK_SET:
   438 |             self.buffer.seek(0)
   439 |             self._traverse_codepoints(0, pos)
   440 |             self._tell = pos
   441 |         # Seek to new position relative to current position
   442 |         elif mode == os.SEEK_CUR:
   443 |             start_pos = self.tell()
   444 |             self._traverse_codepoints(self.tell(), pos)
   445 |             self._tell = start_pos + pos
   446 |         elif mode == os.SEEK_END:
   447 |             self.buffer.seek(0)
   448 |             dest_position = self.len - pos
   449 |             self._traverse_codepoints(0, dest_position)
   450 |             self._tell = dest_position
   451 |         else:
   452 |             raise ValueError(
   453 |                 f"Invalid whence ({mode}, should be 0, 1, or 2)"
   454 |             )
   455 |         return self.tell()

   460 |         self._tell = self.tell() + len(ret)

   525 | class MultiFileReader:
   526 |     """Takes a list of open files or file-like objects and provides an
   527 |     interface to read from them all contiguously. Like
   528 |     :func:`itertools.chain()`, but for reading files.
   529 | 
   530 |        >>> mfr = MultiFileReader(BytesIO(b'ab'), BytesIO(b'cd'), BytesIO(b'e'))
   531 |        >>> mfr.read(3).decode('ascii')
   532 |        u'abc'
   533 |        >>> mfr.read(3).decode('ascii')
   534 |        u'de'
   535 | 
   536 |     The constructor takes as many fileobjs as you hand it, and will
   537 |     raise a TypeError on non-file-like objects. A ValueError is raised
   538 |     when file-like objects are a mix of bytes- and text-handling
   539 |     objects (for instance, BytesIO and StringIO).
   540 |     """
   541 | 
   542 |     def __init__(self, *fileobjs):
   543 |         if not all([callable(getattr(f, 'read', None)) and
   544 |                     callable(getattr(f, 'seek', None)) for f in fileobjs]):
   545 |             raise TypeError('MultiFileReader expected file-like objects'
   546 |                             ' with .read() and .seek()')
   547 |         if all([is_text_fileobj(f) for f in fileobjs]):
   548 |             # codecs.open and io.TextIOBase
   549 |             self._joiner = ''
   550 |         elif any([is_text_fileobj(f) for f in fileobjs]):
   551 |             raise ValueError('All arguments to MultiFileReader must handle'
   552 |                              ' bytes OR text, not a mix')
   553 |         else:
   554 |             # open/file and io.BytesIO
   555 |             self._joiner = b''
   556 |         self._fileobjs = fileobjs
   557 |         self._index = 0
   558 | 
   559 |     def read(self, amt=None):
   560 |         """Read up to the specified *amt*, seamlessly bridging across
   561 |         files. Returns the appropriate type of string (bytes or text)
   562 |         for the input, and returns an empty string when the files are
   563 |         exhausted.
   564 |         """
   565 |         if not amt:
   566 |             return self._joiner.join(f.read() for f in self._fileobjs)
   567 |         parts = []
   568 |         while amt > 0 and self._index < len(self._fileobjs):
   569 |             parts.append(self._fileobjs[self._index].read(amt))
   570 |             got = len(parts[-1])
   571 |             if got < amt:
   572 |                 self._index += 1
   573 |             amt -= got
   574 |         return self._joiner.join(parts)
   575 | 
   576 |     def seek(self, offset, whence=os.SEEK_SET):

   583 |         if offset != 0:
   584 |             raise NotImplementedError(
   585 |                 'MultiFileReader only supports seeking to start at this time')
   586 |         for f in self._fileobjs:
