def test_simple_multiline_with_empty_cells_headerless():
    "Output: simple with multiline cells and empty cells without headers"
    table = [["0", "", ""], ["1", "", ""], ["2", "very long data", "fold\nthis"]]
    expected = "\n".join(
        [
            "-  --------------  ----",
            "0",
            "1",
            "2  very long data  fold",
            "                   this",
            "-  --------------  ----",
        ]
    )
    result = tabulate(table, tablefmt="simple")
    assert_equal(expected, result)

def test_github_multiline():
    "Output: github with multiline cells with headers"
    table = [[2, "foo\nbar"]]
    headers = ("more\nspam eggs", "more spam\n& eggs")
    expected = "\n".join(
        [
            "|        more | more spam   |",
            "|   spam eggs | & eggs      |",
            "|------------:|:------------|",
            "|           2 | foo         |",
            "|             | bar         |",
        ]
    )
    result = tabulate(table, headers, tablefmt="github")
    assert_equal(expected, result)


def test_github_with_colalign():
    "Output: github with explicit column alignment"
    expected = "\n".join(
        [
            "| Name   |   Age |",
            "|:-------|------:|",
            "| Alice  |    24 |",
            "| Bob    |    19 |",
        ]
    )
    result = tabulate(
        [["Alice", 24], ["Bob", 19]],
        ["Name", "Age"],
        tablefmt="github",
        colalign=("left", "right"),
    )
    assert_equal(expected, result)


def test_github_no_alignment():
